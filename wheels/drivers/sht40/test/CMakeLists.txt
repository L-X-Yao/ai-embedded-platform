cmake_minimum_required(VERSION 3.13)
project(test_sht40 C)

# CMAKE_C_STANDARD set as a variable (not target property) intentionally:
# standalone wheel test projects own the entire build, so a top-level variable
# is simpler and consistent across all 140+ wheels.
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)

add_executable(test_sht40
    test_sht40.c
    ../sht40.c
)

target_include_directories(test_sht40 PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/..
    # Convention: all wheel tests live at wheels/<layer>/<name>/test/,
    # so ../../../../tooling/test always resolves to the repo-root tooling dir.
    ${CMAKE_CURRENT_SOURCE_DIR}/../../../../tooling/test
)
target_compile_options(test_sht40 PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion)

# --- null parameter tests (auto-generated by gen_null_tests.py) ---
add_executable(test_null_sht40
    test_null_sht40.c
    ../sht40.c
)
target_include_directories(test_null_sht40 PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/..
    ${CMAKE_CURRENT_SOURCE_DIR}/../../../../tooling/test
)
target_compile_options(test_null_sht40 PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion)

# --- hardware model tests ---
add_executable(test_hw_model_sht40
    test_hw_model_sht40.c
    ../sht40.c
)
target_include_directories(test_hw_model_sht40 PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/..
    ${CMAKE_CURRENT_SOURCE_DIR}/../../../../tooling/test
)
target_compile_options(test_hw_model_sht40 PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion)

enable_testing()
add_test(NAME sht40_tests COMMAND test_sht40)
add_test(NAME sht40_null_tests COMMAND test_null_sht40)
add_test(NAME sht40_hw_model_tests COMMAND test_hw_model_sht40)

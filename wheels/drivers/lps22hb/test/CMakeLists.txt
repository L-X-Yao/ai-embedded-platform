cmake_minimum_required(VERSION 3.13)
project(test_lps22hb C)

# CMAKE_C_STANDARD set as a variable (not target property) intentionally:
# standalone wheel test projects own the entire build, so a top-level variable
# is simpler and consistent across all 140+ wheels.
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)

add_executable(test_lps22hb
    test_lps22hb.c
    ../lps22hb.c
)

target_include_directories(test_lps22hb PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/..
    # Convention: all wheel tests live at wheels/<layer>/<name>/test/,
    # so ../../../../tooling/harness always resolves to the repo-root tooling dir.
    ${CMAKE_CURRENT_SOURCE_DIR}/../../../../tooling/harness
)
target_compile_options(test_lps22hb PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion)

# Null-parameter tests (auto-generated by gen_null_tests.py)
add_executable(test_null_lps22hb
    test_null_lps22hb.c
    ../lps22hb.c
)
target_include_directories(test_null_lps22hb PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/..
    ${CMAKE_CURRENT_SOURCE_DIR}/../../../../tooling/harness
)
target_compile_options(test_null_lps22hb PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion)

# Hardware model tests
add_executable(test_hw_model_lps22hb
    test_hw_model_lps22hb.c
    ../lps22hb.c
)
target_include_directories(test_hw_model_lps22hb PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/..
    ${CMAKE_CURRENT_SOURCE_DIR}/../../../../tooling/harness
)
target_compile_options(test_hw_model_lps22hb PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion)


add_executable(test_boundary_lps22hb
    test_boundary_lps22hb.c
    ../lps22hb.c
)
target_include_directories(test_boundary_lps22hb PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/..
    ${CMAKE_CURRENT_SOURCE_DIR}/../../../../tooling/harness
)
target_compile_options(test_boundary_lps22hb PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion)
enable_testing()
add_test(NAME lps22hb_tests COMMAND test_lps22hb)
add_test(NAME lps22hb_null_tests COMMAND test_null_lps22hb)
add_test(NAME lps22hb_hw_model_tests COMMAND test_hw_model_lps22hb)
add_test(NAME lps22hb_boundary_tests COMMAND test_boundary_lps22hb)
